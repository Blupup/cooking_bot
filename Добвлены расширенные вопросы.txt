    def get_updates(self):
        """Получение обновлений от Telegram"""
        try:
            response = requests.get(
                f"{self.base_url}/getUpdates",
                params={
                    "offset": self.last_update_id + 1,
                    "timeout": 10,
                    "limit": 10
                },
                timeout=5
            )
            return response.json().get("result", []) if response.status_code == 200 else []
        except:
            return []

    def send_message(self, chat_id, text, inline_buttons=None, persistent_keyboard=False):
        """Отправка сообщений"""
        def _send():
            try:
                payload = {
                    "chat_id": chat_id,
                    "text": text,
                    "parse_mode": "HTML",
                    "disable_web_page_preview": True
                }
                
                if inline_buttons:
                    payload["reply_markup"] = json.dumps({"inline_keyboard": inline_buttons})
                elif persistent_keyboard:
                    payload["reply_markup"] = json.dumps(PERSISTENT_KEYBOARD)
                
                requests.post(f"{self.base_url}/sendMessage", json=payload, timeout=3)
            except:
                pass
        
        threading.Thread(target=_send, daemon=True).start()

    def determine_prompt_type(self, user_input):
        """Определяет тип запроса для лучшего промта"""
        text_lower = user_input.lower()
        
        # Ключевые слова для разных типов запросов
        holiday_keywords = ['праздник', 'день рождения', 'новый год', 'романтич', 'ужин для двоих', 'гости', 'торжеств', 'юбилей']
        diet_keywords = ['вегетариан', 'низкокалорий', 'диетич', 'без глютена', 'безлактоз', 'здоров', 'полезн', 'похуден']
        quick_keywords = ['быстр', 'за 15', 'за 20', 'за 30', 'срочно', 'экспресс', 'минут', 'быстро', 'сразу']
        cuisine_keywords = ['итальянск', 'японск', 'мексикан', 'русск', 'французск', 'китайск', 'грузинск', 'восточн']
        
        if any(keyword in text_lower for keyword in holiday_keywords):
            return "holiday"
        elif any(keyword in text_lower for keyword in diet_keywords):
            return "diet"
        elif any(keyword in text_lower for keyword in quick_keywords):
            return "quick"
        elif any(keyword in text_lower for keyword in cuisine_keywords):
            return "cuisine"
        elif len(user_input.split()) <= 3 and not any(word in text_lower for word in ['из', 'с', 'и', 'или']):
            return "dish_name"
        else:
            return "products"

    def process_message(self, message):
        """Обработка входящих сообщений"""
        chat_id = message["chat"]["id"]
        text = message.get("text", "").strip()

        # Обработка команд
        if text in ["/start", "/help", "🏠 Главное меню"]:
            self.send_message(chat_id, WELCOME_TEXT, persistent_keyboard=True)
            return

        if text == "🔄 Новый запрос":
            self.send_message(chat_id, "📝 Что готовим? Введите продукты или название блюда:", 
                            persistent_keyboard=True)
            return

        if text == "🔍 Возможности бота":
            self.send_message(chat_id, BOT_CAPABILITIES, persistent_keyboard=True)
            return

        if not text:
            self.send_message(chat_id, "✍️ Напишите что приготовить...", persistent_keyboard=True)
            return

        # Умная проверка запроса
        is_valid, error_msg, suggestion = self.smart_validation(text)
        
        if not is_valid:
            full_message = error_msg
            if suggestion:
                full_message += f"\n\n{suggestion}"
            self.send_message(chat_id, full_message, persistent_keyboard=True)
            return

        # Сразу отправляем сообщение о подборе рецепта
        self.send_message(chat_id, "⚡ Подбираю рецепт...")

        # Обработка рецепта в отдельном потоке
        def process_recipe():
            start_time = time.time()
            ai_text = self.get_ai_response(text)
            end_time = time.time()
            print(f"✅ Рецепт готов за {end_time - start_time:.1f} сек")
            
            # Отправляем готовый рецепт
            self.send_message(chat_id, ai_text, 
                            inline_buttons=self.build_inline_buttons(), 
                            persistent_keyboard=True)

        self.thread_pool.submit(process_recipe)

    def format_text(self, text):
        """Форматирование текста рецепта"""
        # Замена заголовков
        replacements = {
            "Ингредиенты:": "🥕 <b>Ингредиенты:</b>",
            "Приготовление:": "👨‍🍳 <b>Приготовление:</b>",
            "Время:": "⏱️ <b>Время:</b>",
            "Совет:": "💡 <b>Совет:</b>",
            "Советы:": "💡 <b>Советы:</b>",
            "Калорийность:": "🔥 <b>Калорийность:</b>",
            "Сложность:": "🎯 <b>Сложность:</b>",
            "Польза:": "💪 <b>Польза:</b>",
            "На персон:": "👥 <b>На персон:</b>",
            "Пищевая ценность:": "📊 <b>Пищевая ценность:</b>",
            "Шаги:": "👨‍🍳 <b>Шаги:</b>",
            "Инструкция:": "📝 <b>Инструкция:</b>",
            "Рецепт:": "🍳 <b>Рецепт:</b>",
            "Особенности:": "✨ <b>Особенности:</b>",
            "Сервировка:": "🍽️ <b>Сервировка:</b>"
        }
        
        for old, new in replacements.items():
            text = text.replace(old, new)
        
        # Разметка жирным
        text = re.sub(r"\*\*(.*?)\*\*", r"<b>\1</b>", text)
        text = re.sub(r"\*(.*?)\*", r"<i>\1</i>", text)
        
        # Добавление эмодзи к спискам
        lines = text.split('\n')
        formatted_lines = []
        for line in lines:
            stripped = line.strip()
            if stripped.startswith('•') or stripped.startswith('-') or (stripped and stripped[0].isdigit() and '.' in stripped[:3]):
                formatted_lines.append(line)
            else:
                formatted_lines.append(line)
        
        text = '\n'.join(formatted_lines)
        
        # Ограничение длины
        if len(text) > 2000:
            text = text[:2000] + "\n\n... (рецепт сокращен для удобства)"
        
        return text.strip()

    def build_inline_buttons(self):
        """Создание inline-кнопок"""
        return [[
            {"text": "🏠 Главное меню", "callback_data": "home"},
            {"text": "🔄 Новый рецепт", "callback_data": "restart"},
            {"text": "🔍 Возможности", "callback_data": "capabilities"}
        ]]

    def process_callback(self, callback):
        """Обработка callback-запросов"""
        callback_id = callback.get("id")
        data = callback.get("data", "")
        message = callback.get("message", {})
        chat_id = message["chat"]["id"]

        if data == "home":
            self.send_message(chat_id, WELCOME_TEXT, persistent_keyboard=True)
        elif data == "restart":
            self.send_message(chat_id, "📝 Что готовим дальше?", persistent_keyboard=True)
        elif data == "capabilities":
            self.send_message(chat_id, BOT_CAPABILITIES, persistent_keyboard=True)
        
        # Ответ на callback
        try:
            requests.post(f"{self.base_url}/answerCallbackQuery", 
                         data={"callback_query_id": callback_id, "text": "✅"}, 
                         timeout=1)
        except:
            pass

        """Запуск бота"""
        print("""
⚡ УМНЫЙ КУЛИНАРНЫЙ БОТ ЗАПУЩЕН!
        
🎯 ОСНОВНЫЕ ФУНКЦИИ:

• "праздничное меню на НГ"
• "быстрый завтрак за 10 минут"
• и многое другое!
        """)
        
        while True:
            try:
                updates = self.get_updates()
                for update in updates:
                    self.last_update_id = update.get("update_id", self.last_update_id)
                    
                    if "message" in update and "text" in update["message"]:
                        self.process_message(update["message"])
                    elif "callback_query" in update:
                        self.process_callback(update["callback_query"])
                        
                time.sleep(0.3)
                
            except KeyboardInterrupt:
                print("\n👋 Выключение бота...")
                self.thread_pool.shutdown()
                break
            except Exception as e:
                print(f"⚠️ Ошибка: {e}")
                time.sleep(1)
