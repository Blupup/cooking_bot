import time
import json
import re
import requests
from concurrent.futures import ThreadPoolExecutor
import threading

# --- 🔑 КЛЮЧИ ---
TELEGRAM_TOKEN = "-"
OPENROUTER_API_KEY = "-"

# --- 📋 ТЕКСТЫ ---
WELCOME_TEXT = """👨‍🍳 <b>Добро пожаловать в Кулинарного Помощника!</b>

Я помогу вам с рецептами:

🍎 <b>Поиск по продуктам</b> — что у вас есть  
🍽️ <b>Рецепт по названию</b> — название блюда

Пример: <i>курица, рис, овощи</i> или <i>борщ</i>"""

# --- 📱 КЛАВИАТУРА ---
PERSISTENT_KEYBOARD = {
    "keyboard": [["🏠 Главное меню", "🔄 Новый запрос"]],
    "resize_keyboard": True,
    "one_time_keyboard": False
}

class FastCookingBot:
    def __init__(self):
        self.base_url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}"
        self.last_update_id = 0
        self.openrouter_url = "https://openrouter.ai/api/v1/chat/completions"
        self.headers = {
            "Authorization": f"Bearer {OPENROUTER_API_KEY}",
            "Content-Type": "application/json",
            "HTTP-Referer": "https://yourapp.example",
            "X-Title": "CookingBot"
        }
        
        # Кэш для быстрых ответов
        self.recipe_cache = {}
        self.validation_cache = {}  # Кэш проверок
        self.max_cache_size = 50
        
        # Пул потоков для параллельной обработки
        self.thread_pool = ThreadPoolExecutor(max_workers=5)
        
        # Быстрые ответы для частых запросов
        self.quick_responses = {
            "омлет": "🍳 <b>Простой омлет</b>\n\n🥕 <b>Ингредиенты:</b>\n• Яйца - 2 шт.\n• Молоко - 2 ст.л.\n• Соль - щепотка\n• Масло - 1 ч.л.\n\n👨‍🍳 <b>Приготовление:</b>\n1. Взбейте яйца с молоком и солью\n2. Разогрейте сковороду с маслом\n3. Вылейте смесь, жарьте 3-4 минуты\n\n⏱️ <b>Время:</b> 10 минут\n💡 <b>Совет:</b> Добавьте зелень для аромата",
            "салат": "🥗 <b>Свежий салат</b>\n\n🥕 <b>Ингредиенты:</b>\n• Огурцы - 2 шт.\n• Помидоры - 2 шт.\n• Лук - 0.5 шт.\n• Масло, соль\n\n👨‍🍳 <b>Приготовление:</b>\n1. Нарежьте овощи\n2. Добавьте соль, масло\n3. Перемешайте\n\n⏱️ <b>Время:</b> 5 минут\n💡 <b>Совет:</b> Дайте постоять 5 минут",
            "блины": "🥞 <b>Блины классические</b>\n\n🥕 <b>Ингредиенты:</b>\n• Мука - 200г\n• Молоко - 500мл\n• Яйца - 2 шт.\n• Сахар - 2 ст.л.\n• Соль - щепотка\n\n👨‍🍳 <b>Приготовление:</b>\n1. Смешайте все ингредиенты\n2. Дайте тесту постоять 15 мин\n3. Жарьте на разогретой сковороде\n\n⏱️ <b>Время:</b> 30 минут\n💡 <b>Совет:</b> Смазывайте сковороду маслом",
        }

        # Быстрые проверки для частых запросов
        self.quick_validation = {
            # Очевидные пищевые запросы
            "омлет": (True, ""),
            "салат": (True, ""),
            "блины": (True, ""),
            "суп": (True, ""),
            "борщ": (True, ""),
            "каша": (True, ""),
            "кофе": (True, ""),
            "чай": (True, ""),
            # Очевидные несъедобные предметы
            "телефон": (False, "📱 Телефон нельзя приготовить и съесть. Я помогаю только с рецептами еды!"),
            "компьютер": (False, "💻 Компьютер - это техника, а не еда. Попробуйте запросить рецепт из продуктов!"),
            "машина": (False, "🚗 Машину нельзя приготовить! Я специализируюсь на кулинарных рецептах."),
            "дерево": (False, "🌳 Дерево не съедобно! Укажите пищевые продукты для рецепта."),
            "камень": (False, "🪨 Камни не готовят! Я помогу с рецептами из мяса, овощей, круп и других продуктов."),
            "кирпич": (False, "🧱 Кирпич - строительный материал, а не еда! Попробуйте запросить рецепт выпечки или других блюд."),
        }

        # Базовые стоп-слова для быстрой проверки
        self.stop_words = {
            'бомба', 'оружие', 'наркотик', 'убить', 'взрыв', 'алкоголь', 'вино', 'водка', 
            'пиво', 'коньяк', 'виски', 'ром', 'джин', 'текила', 'самогон', 'сигареты'
        }

    def get_updates(self):
        """Получение обновлений от Telegram"""
        try:
            response = requests.get(
                f"{self.base_url}/getUpdates",
                params={
                    "offset": self.last_update_id + 1,
                    "timeout": 10,
                    "limit": 10
                },
                timeout=5
            )
            return response.json().get("result", []) if response.status_code == 200 else []
        except:
            return []

    def send_message(self, chat_id, text, inline_buttons=None, persistent_keyboard=False):
        """Отправка сообщений"""
        def _send():
            try:
                payload = {
                    "chat_id": chat_id,
                    "text": text,
                    "parse_mode": "HTML",
                    "disable_web_page_preview": True
                }
                
                if inline_buttons:
                    payload["reply_markup"] = json.dumps({"inline_keyboard": inline_buttons})
                elif persistent_keyboard:
                    payload["reply_markup"] = json.dumps(PERSISTENT_KEYBOARD)
                
                requests.post(f"{self.base_url}/sendMessage", json=payload, timeout=3)
            except:
                pass
        
        threading.Thread(target=_send, daemon=True).start()

    def validate_with_ai(self, user_input):
        """Умная проверка запроса через AI"""
        cache_key = f"validate_{user_input.lower().strip()}"
        
        # Проверка кэша
        if cache_key in self.validation_cache:
            return self.validation_cache[cache_key]

        prompt = f"""Проанализируй запрос и определи, относится ли он к кулинарии и приготовлению еды.

Запрос: "{user_input}"

Ответь ТОЛЬКО в формате JSON:
{{
    "is_food_related": true/false,
    "reason": "краткое объяснение",
    "suggestion": "предложение что делать дальше"
}}

Правила:
- Считай пищевыми: продукты, ингредиенты, названия блюд, кулинарные процессы
- Несъедобное: техника, одежда, мебель, животные, растения, предметы
- Если сомневаешься - считай что это еда"""

        data = {
            "model": "deepseek/deepseek-chat",
            "messages": [
                {
                    "role": "system",
                    "content": "Ты аналитик запросов. Отвечай ТОЛЬКО в формате JSON. Будь строг к несъедобным предметам, но снисходителен к возможным названиям блюд."
                },
                {
                    "role": "user", 
                    "content": prompt
                }
            ],
            "temperature": 0.3,
            "max_tokens": 150,
            "stream": False
        }

        try:
            response = requests.post(
                self.openrouter_url,
                headers=self.headers,
                json=data,
                timeout=4  # Быстрая проверка
            )
            
            if response.status_code == 200:
                result = response.json()
                text = result['choices'][0]['message']['content'].strip()
                
                # Парсинг JSON ответа
                try:
                    validation_data = json.loads(text)
                    is_valid = validation_data.get("is_food_related", True)  # По умолчанию разрешаем
                    reason = validation_data.get("reason", "")
                    suggestion = validation_data.get("suggestion", "")
                    
                    result = (is_valid, reason, suggestion)
                    
                    # Кэшируем результат
                    if len(self.validation_cache) >= self.max_cache_size:
                        self.validation_cache.pop(next(iter(self.validation_cache)))
                    self.validation_cache[cache_key] = result
                    
                    return result
                    
                except json.JSONDecodeError:
                    # Если AI вернул не JSON, разрешаем запрос (безопасный подход)
                    return (True, "", "Продолжаем поиск рецепта")
                    
            else:
                return (True, "", "Продолжаем поиск рецепта")  # При ошибке разрешаем
                
        except Exception:
            return (True, "", "Продолжаем поиск рецепта")  # При ошибке разрешаем

    def smart_validation(self, text):
        """Умная многоуровневая проверка запроса"""
        text_lower = text.lower().strip()
        
        # 1. Быстрая проверка стоп-слов
        if any(stop_word in text_lower for stop_word in self.stop_words):
            return False, "🚫 Этот запрос содержит запрещенные слова", "Пожалуйста, укажите пищевые продукты"

        # 2. Проверка быстрых ответов
        if text_lower in self.quick_validation:
            is_valid, message = self.quick_validation[text_lower]
            if not is_valid:
                return False, message, "Попробуйте запросить рецепт из продуктов питания"
            return True, "", ""

        # 3. Проверка очевидных паттернов
        # Слишком длинные запросы (возможно, не о еде)
        if len(text) > 60:
            return False, "📝 Запрос слишком сложный для анализа", "Укажите конкретные продукты или название блюда"

        # Запросы с числами и единицами измерения (скорее всего о еде)
        if re.search(r'\d+\s*(г|кг|мл|л|шт|стакан|ст\.?л|ч\.?л)', text_lower):
            return True, "", ""

        # 4. AI-проверка для сложных случаев
        is_valid, reason, suggestion = self.validate_with_ai(text)
        
        if not is_valid:
            error_msg = f"🚫 {reason}" if reason else "🚫 Этот запрос не относится к кулинарии"
            suggestion_msg = suggestion if suggestion else "Попробуйте указать продукты питания"
            return False, error_msg, suggestion_msg
        
        return True, "", ""

    def get_ai_response(self, user_input):
        """Генерация рецептов через AI"""
        cache_key = user_input.lower().strip()
        
        if cache_key in self.recipe_cache:
            return self.recipe_cache[cache_key]
        
        if cache_key in self.quick_responses:
            return self.quick_responses[cache_key]

        prompt = f"""Создай краткий рецепт для: {user_input}

Формат:
Название блюда
Ингредиенты списком
Краткие шаги приготовления
Время готовки
Полезный совет

Максимально кратко и понятно:"""

        data = {
            "model": "deepseek/deepseek-chat",
            "messages": [
                {
                    "role": "system",
                    "content": "Ты шеф-повар. Отвечай очень кратко, используй эмодзи. Только рецепты. Формат: название, ингредиенты списком, шаги кратко, время, совет. Не более 300 символов."
                },
                {
                    "role": "user", 
                    "content": prompt
                }
            ],
            "temperature": 0.7,
            "max_tokens": 250,
            "stream": False
        }

        try:
            response = requests.post(
                self.openrouter_url,
                headers=self.headers,
                json=data,
                timeout=6
            )
            
            if response.status_code == 200:
                result = response.json()
                text = result['choices'][0]['message']['content'].strip()
                formatted_text = self.format_text(text)
                
                if len(self.recipe_cache) >= self.max_cache_size:
                    self.recipe_cache.pop(next(iter(self.recipe_cache)))
                self.recipe_cache[cache_key] = formatted_text
                
                return formatted_text
            else:
                return self.get_fallback_recipe(user_input)
                
        except Exception:
            return self.get_fallback_recipe(user_input)

    def get_fallback_recipe(self, user_input):
        """Резервный рецепт при ошибках"""
        return f"""🍳 <b>Быстрый рецепт из {user_input}</b>

🥕 <b>Ингредиенты:</b>
• {user_input}
• Соль, перец
• Масло растительное

👨‍🍳 <b>Приготовление:</b>
1. Подготовьте {user_input}
2. Обжарьте или отварите
3. Добавьте специи
4. Подавайте!

⏱️ <b>Время:</b> 20 минут
💡 <b>Совет:</b> Экспериментируйте со специями"""

    def process_message(self, message):
        """Обработка входящих сообщений"""
        chat_id = message["chat"]["id"]
        text = message.get("text", "").strip()

        # Обработка команд
        if text in ["/start", "/help", "🏠 Главное меню"]:
            self.send_message(chat_id, WELCOME_TEXT, persistent_keyboard=True)
            return

        if text == "🔄 Новый запрос":
            self.send_message(chat_id, "📝 Что готовим? Введите продукты или название блюда:", persistent_keyboard=True)
            return

        if not text:
            self.send_message(chat_id, "✍️ Напишите что приготовить...", persistent_keyboard=True)
            return

        # Умная проверка запроса
        is_valid, error_msg, suggestion = self.smart_validation(text)
        
        if not is_valid:
            full_message = error_msg
            if suggestion:
                full_message += f"\n\n💡 {suggestion}"
            self.send_message(chat_id, full_message, persistent_keyboard=True)
            return

        # Сразу отправляем сообщение о подборе рецепта
        self.send_message(chat_id, "⚡ Подбираю рецепт...")

        # Обработка рецепта в отдельном потоке
        def process_recipe():
            start_time = time.time()
            ai_text = self.get_ai_response(text)
            end_time = time.time()
            print(f"✅ Рецепт готов за {end_time - start_time:.1f} сек")
            
            # Отправляем готовый рецепт
            self.send_message(chat_id, ai_text, 
                            inline_buttons=self.build_inline_buttons(), 
                            persistent_keyboard=True)

        self.thread_pool.submit(process_recipe)

    def format_text(self, text):
        """Форматирование текста рецепта"""
        text = text.replace("Ингредиенты:", "🥕 <b>Ингредиенты:</b>")
        text = text.replace("Приготовление:", "👨‍🍳 <b>Приготовление:</b>")
        text = text.replace("Время:", "⏱️ <b>Время:</b>")
        text = text.replace("Совет:", "💡 <b>Совет:</b>")
        
        text = re.sub(r"\*\*(.*?)\*\*", r"<b>\1</b>", text)
        text = re.sub(r"\*(.*?)\*", r"<i>\1</i>", text)
        
        return text.strip()

    def build_inline_buttons(self):
        """Создание inline-кнопок"""
        return [[
            {"text": "🏠 Главное меню", "callback_data": "home"},
            {"text": "🔄 Новый рецепт", "callback_data": "restart"}
        ]]

    def process_callback(self, callback):
        """Обработка callback-запросов"""
        callback_id = callback.get("id")
        data = callback.get("data", "")
        message = callback.get("message", {})
        chat_id = message["chat"]["id"]

        if data == "home":
            self.send_message(chat_id, WELCOME_TEXT, persistent_keyboard=True)
        elif data == "restart":
            self.send_message(chat_id, "📝 Что готовим дальше?", persistent_keyboard=True)
        
        # Ответ на callback
        try:
            requests.post(f"{self.base_url}/answerCallbackQuery", 
                         data={"callback_query_id": callback_id, "text": "✅"}, 
                         timeout=1)
        except:
            pass

    def run(self):
        """Запуск бота"""
        print("⚡ УМНЫЙ КУЛИНАРНЫЙ БОТ ЗАПУЩЕН!")
        print("🧠 Умная проверка: AI-анализ запросов + быстрые фильтры")
        print("🎯 Оптимизации: кэш, быстрые ответы, потоки")
        print("⏱️  Цель: ответ за 2-5 секунд")
        
        while True:
            try:
                updates = self.get_updates()
                for update in updates:
                    self.last_update_id = update.get("update_id", self.last_update_id)
                    
                    if "message" in update and "text" in update["message"]:
                        self.process_message(update["message"])
                    elif "callback_query" in update:
                        self.process_callback(update["callback_query"])
                        
                time.sleep(0.3)
                
            except KeyboardInterrupt:
                print("\n👋 Выключение...")
                self.thread_pool.shutdown()
                break
            except Exception as e:
                print(f"Ошибка: {e}")
                time.sleep(1)

if __name__ == "__main__":
    bot = FastCookingBot()

    bot.run()
